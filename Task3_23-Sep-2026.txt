Task 3 – List Rendering & Conditional Rendering

1. App.jsx
===========
// App.jsx
import React, { useState } from 'react';
import Home from './Home';
import Login from './Login';
import { styles } from './styles';

export default function App() {
  // Boolean state for conditional rendering
  const [isLoggedIn, setIsLoggedIn] = useState(false);

  const handleLogin = () => setIsLoggedIn(true);
  const handleLogout = () => setIsLoggedIn(false);

  return (
    <div style={styles.appWrapper}>
      {/* Conditional Rendering using Ternary Operator */}
      {isLoggedIn ? (
        <Home onLogout={handleLogout} />
      ) : (
        <Login onLogin={handleLogin} />
      )}
    </div>
  );
}

2. Home.jsx

import React from 'react';
import { mockUsers } from './userData'; // Changed from '../data/userData'
import { styles } from './styles';       // Changed from '../styles/styles'

export default function Home({ onLogout }) {
  return (
    <div style={styles.homeContainer}>
      <header style={styles.header}>
        <div>
          <h1 style={styles.heading}>Welcome Back!</h1>
          <p style={styles.subheading}>Team Directory Overview</p>
        </div>
        <button style={styles.logoutBtn} onClick={onLogout}>
          Log Out
        </button>
      </header>

      {/* Dynamic List Rendering using map() */}
      <div style={styles.gridContainer}>
        {mockUsers.map((user) => (
          <div key={user.id} style={styles.card}>
            <div style={styles.badge}>{user.role}</div>
            <h3 style={styles.cardTitle}>{user.name}</h3>
            <div style={styles.cardBody}>
              <p style={styles.cardText}>
                <strong>Email:</strong> {user.email}
              </p>
              <p style={styles.cardText}>
                <strong>Age:</strong> {user.age}
              </p>
              <p style={styles.cardText}>
                <strong>City:</strong> {user.city}
              </p>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
