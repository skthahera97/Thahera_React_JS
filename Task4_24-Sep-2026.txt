Task 4 – Implement useState Hook

1. App.jsx
==========
import { useState } from "react";

function App() {
  // Counter state
  const [count, setCount] = useState(0);

  // Theme state
  const [isDark, setIsDark] = useState(false);

  // Counter functions
  const increment = () => {
    setCount(count + 1);
  };

  const decrement = () => {
    setCount(count - 1);
  };

  const reset = () => {
    setCount(0);
  };

  // Theme toggle
  const toggleTheme = () => {
    setIsDark(!isDark);
  };

  return (
    <div className={isDark ? "app dark" : "app light"}>

      {/* Header */}
      <div className="header">
        <h1>React Demo</h1>

        <button className="theme-btn" onClick={toggleTheme}>
          {isDark ? "☀️ Light Theme" : "🌙 Dark Theme"}
        </button>
      </div>

      {/* Main Container */}
      <div className="container">

        {/* Counter Card */}
        <div className="card">
          <h2>Counter</h2>

          <div className="count">
            {count}
          </div>

          <div className="button-group">
            <button
              className="btn increment"
              onClick={increment}
            >
              + Increment
            </button>

            <button
              className="btn decrement"
              onClick={decrement}
            >
              - Decrement
            </button>

            <button
              className="btn reset"
              onClick={reset}
            >
              ↻ Reset
            </button>
          </div>
        </div>

        {/* Theme Card */}
        <div className="card theme-card">
          <h2>Theme Toggle</h2>

          <p>
            Current Theme:
            <strong> {isDark ? "Dark" : "Light"}</strong>
          </p>

          <button
            className="toggle-btn"
            onClick={toggleTheme}
          >
            {isDark
              ? "☀️ Switch to Light Theme"
              : "🌙 Switch to Dark Theme"}
          </button>
        </div>

      </div>

      {/* Footer */}
      <footer>
        <p>Built with React useState Hook</p>
      </footer>

    </div>
  );
}

export default App;
