Task 11 – Implement useSearchParams Hook with Pagination

1. App.jsx
==========
import React from 'react'
import { useSearchParams } from 'react-router-dom'
import users from './data.js'

export default function App() {
  const [searchParams, setSearchParams] = useSearchParams()

  // Items per page
  const limit = 5

  // Get current page from URL query string (default to 1)
  const page = parseInt(searchParams.get('page') || '1', 10)

  // Calculate total pages and slice current page data
  const totalPages = Math.ceil(users.length / limit)
  const startIndex = (page - 1) * limit
  const currentUsers = users.slice(startIndex, startIndex + limit)

  return (
    <div className="container">
      <h1>User List</h1>
      <p className="subtitle">Pagination with useSearchParams</p>

      {/* User Table */}
      <div className="table-container">
        <table>
          <thead>
            <tr>
              <th>ID</th>
              <th>Name</th>
              <th>Email</th>
              <th>Age</th>
              <th>City</th>
            </tr>
          </thead>
          <tbody>
            {currentUsers.map((user) => (
              <tr key={user.id}>
                <td>{user.id}</td>
                <td>{user.name}</td>
                <td>{user.email}</td>
                <td>{user.age}</td>
                <td>{user.city}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* Pagination Controls */}
      <div className="pagination">
        <button 
          disabled={page <= 1} 
          onClick={() => setSearchParams({ page: (page - 1).toString() })}
        >
          Prev
        </button>
        <span>
          Page {page} of {totalPages}
        </span>
        <button 
          disabled={page >= totalPages}
          onClick={() => setSearchParams({ page: (page + 1).toString() })}
        >
          Next
        </button>
      </div>

      <div className="url-info">
        Current Page Query Parameter: <strong>Page={page}</strong>
      </div>
    </div>
  )
}
